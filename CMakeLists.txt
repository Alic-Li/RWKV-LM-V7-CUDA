cmake_minimum_required(VERSION 3.24)
option(RWKV_ROCM "Build the executables with ROCm instead of CUDA" OFF)

if(RWKV_ROCM)
  project(rwkv7_pretrain LANGUAGES CXX HIP)
  find_package(hip REQUIRED CONFIG)
  find_package(rocblas REQUIRED CONFIG)
  find_package(hipsolver REQUIRED CONFIG)
  find_path(HIPCUB_INCLUDE_DIR hipcub/device/device_radix_sort.hpp
    HINTS "${CMAKE_HIP_COMPILER_ROCM_ROOT}" PATH_SUFFIXES include REQUIRED)
else()
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
  endif()
  project(rwkv7_pretrain LANGUAGES CXX CUDA)
  find_package(CUDAToolkit 12 REQUIRED)
endif()
find_package(LibArchive REQUIRED)
option(RWKV_NCCL "Enable NCCL or RCCL data parallel training when available" ON)

add_library(rwkv_pth STATIC
  pretrain/weights.cpp pretrain/compat/pth_archive.cpp pretrain/compat/pth_tensor.cpp)
target_compile_features(rwkv_pth PUBLIC cxx_std_17)
target_include_directories(rwkv_pth PUBLIC pretrain/compat)
target_link_libraries(rwkv_pth PRIVATE LibArchive::LibArchive)

set(RWKV_GPU_SOURCES pretrain/model.cu
  cuda/gradient_norm.cu cuda/embedding.cu cuda/adamw.cu cuda/value_bias.cu cuda/layer_norm.cu)
add_executable(rwkv7_init pretrain/init.cu)
add_executable(rwkv7_train pretrain/train.cpp pretrain/dataset.cpp pretrain/compat/jsonl.cpp
  ${RWKV_GPU_SOURCES})
target_include_directories(rwkv7_train PRIVATE cuda pretrain)
target_compile_definitions(rwkv7_train PRIVATE _N_=64 _CHUNK_LEN_=16)
target_link_libraries(rwkv7_init PRIVATE rwkv_pth)
target_link_libraries(rwkv7_train PRIVATE rwkv_pth)

if(RWKV_ROCM)
  add_subdirectory(hip)
  set_source_files_properties(pretrain/init.cu ${RWKV_GPU_SOURCES} PROPERTIES LANGUAGE HIP)
  set_target_properties(rwkv7_init rwkv7_train PROPERTIES HIP_STANDARD 17 HIP_STANDARD_REQUIRED YES)
  foreach(target rwkv7_init rwkv7_train)
    target_compile_definitions(${target} PRIVATE RWKV_ROCM)
    target_include_directories(${target} PRIVATE hip/compat)
  endforeach()
  target_link_libraries(rwkv7_init PRIVATE hip::host roc::hipsolver)
  target_include_directories(rwkv7_train PRIVATE "${HIPCUB_INCLUDE_DIR}")
  target_link_libraries(rwkv7_train PRIVATE hip::host roc::rocblas rwkv7_hip_ops)
  if(RWKV_NCCL)
    find_package(rccl QUIET CONFIG)
    if(TARGET roc::rccl)
      target_link_libraries(rwkv7_train PRIVATE roc::rccl)
      target_compile_definitions(rwkv7_train PRIVATE RWKV_HAVE_NCCL)
    else()
      message(STATUS "RCCL unavailable; building single-GPU training")
    endif()
  endif()
else()
  target_sources(rwkv7_train PRIVATE
    cuda/rwkv7_clampw.cu cuda/rwkv7_clampw_v3_for_h100.cu
    cuda/rwkv7_cmix_bf16_v5.cu cuda/rwkv7_head_l2wrap_ce_bf16_v4.cu
    cuda/rwkv7_tmix_a_gate_bf16.cu cuda/rwkv7_tmix_kk_pre_bf16_v5.cu
    cuda/rwkv7_tmix_lnx_rkvres_xg_bf16_v1.cu cuda/rwkv7_tmix_mix6_bf16_v5.cu
    cuda/rwkv7_tmix_vres_gate_bf16_v1.cu)
  target_compile_options(rwkv7_train PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--use_fast_math>)
  # Match ATen division, square root and denormal handling in these operators.
  set_source_files_properties(${RWKV_GPU_SOURCES} PROPERTIES
    COMPILE_OPTIONS "--prec-div=true;--prec-sqrt=true;--ftz=false")
  target_link_libraries(rwkv7_init PRIVATE CUDA::cusolver)
  target_link_libraries(rwkv7_train PRIVATE CUDA::cudart CUDA::cublas)
  if(RWKV_NCCL)
    find_path(NCCL_INCLUDE_DIR nccl.h)
    find_library(NCCL_LIBRARY nccl)
    if(NCCL_INCLUDE_DIR AND NCCL_LIBRARY)
      target_include_directories(rwkv7_train PRIVATE "${NCCL_INCLUDE_DIR}")
      target_link_libraries(rwkv7_train PRIVATE "${NCCL_LIBRARY}")
      target_compile_definitions(rwkv7_train PRIVATE RWKV_HAVE_NCCL)
    else()
      message(STATUS "NCCL unavailable; building single-GPU training")
    endif()
  endif()
  set_target_properties(rwkv7_init rwkv7_train PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
endif()
